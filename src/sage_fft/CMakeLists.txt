cmake_minimum_required(VERSION 3.16)
project(sage_fft LANGUAGES CXX)
set(RUN_MODE "npu" CACHE STRING "run mode")
set(SOC_VERSION "Ascend310P1" CACHE STRING "soc")
if(DEFINED ENV{ASCEND_CANN_PACKAGE_PATH})
  set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_CANN_PACKAGE_PATH}" CACHE PATH "CANN toolkit")
else()
  set(ASCEND_CANN_PACKAGE_PATH "/usr/local/Ascend/ascend-toolkit/latest" CACHE PATH "CANN toolkit")
endif()
set(ASCENDC_CMAKE_DIR ${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake)
include(${ASCENDC_CMAKE_DIR}/ascendc.cmake)
ascendc_library(sage_kernels_npu STATIC kernels.cpp)
add_executable(sage_fft main.cpp)
target_compile_options(sage_fft PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-std=c++17>)
target_include_directories(sage_fft PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${ASCEND_CANN_PACKAGE_PATH}/runtime/include)
target_link_libraries(sage_fft PRIVATE $<BUILD_INTERFACE:host_intf_pub> sage_kernels_npu ascendcl c_sec runtime stdc++ m)
