cmake_minimum_required(VERSION 3.18)
project(sage_native_fft LANGUAGES CXX)
set(RUN_MODE "npu" CACHE STRING "AscendC run mode")
set(SOC_VERSION "Ascend310P1" CACHE STRING "Target Ascend SoC; this source was measured on Ascend310P1")
if(NOT ASCEND_CANN_PACKAGE_PATH)
  if(DEFINED ENV{ASCEND_CANN_PACKAGE_PATH})
    set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_CANN_PACKAGE_PATH}" CACHE PATH "CANN toolkit root")
  elseif(DEFINED ENV{ASCEND_HOME_PATH})
    set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_HOME_PATH}" CACHE PATH "CANN toolkit root")
  else()
    message(FATAL_ERROR "Set ASCEND_CANN_PACKAGE_PATH or pass -DASCEND_CANN_PACKAGE_PATH=/path/to/toolkit")
  endif()
endif()
set(ASCENDC_CMAKE_DIR "${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
if(NOT EXISTS "${ASCENDC_CMAKE_DIR}/ascendc.cmake")
  message(FATAL_ERROR "CANN toolkit does not contain compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake")
endif()
include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
ascendc_library(sage_kernels_npu STATIC kernels.cpp)
add_executable(sage_fft main.cpp)
target_compile_features(sage_fft PRIVATE cxx_std_17)
target_include_directories(sage_fft PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${ASCEND_CANN_PACKAGE_PATH}/runtime/include")
target_link_libraries(sage_fft PRIVATE $<BUILD_INTERFACE:host_intf_pub> sage_kernels_npu ascendcl c_sec runtime stdc++ m)
